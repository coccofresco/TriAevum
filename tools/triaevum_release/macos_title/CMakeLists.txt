cmake_minimum_required(VERSION 3.26)
project(TriAevumMacTitle LANGUAGES CXX)
if(NOT APPLE OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "arm64|aarch64")
    message(FATAL_ERROR "This title target currently supports macOS Apple Silicon")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
get_filename_component(root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(TRIAEVUM_TRANSLATED_TITLE_DIR "" CACHE PATH "Extracted published translated title source")
set(manifest "${TRIAEVUM_TRANSLATED_TITLE_DIR}/TITLE_SOURCE_MANIFEST.json")
file(READ "${manifest}" metadata)
string(JSON format GET "${metadata}" format)
if(NOT format STREQUAL "triaevum_translated_title_source_v1")
    message(FATAL_ERROR "Unsupported translated source manifest")
endif()
string(JSON count LENGTH "${metadata}" files)
math(EXPR last "${count} - 1")
set(sources)
foreach(index RANGE 0 ${last})
    string(JSON name MEMBER "${metadata}" files ${index})
    if(NOT name MATCHES "^[A-Za-z0-9_]+\\.(cpp|h)$")
        message(FATAL_ERROR "Invalid translated source name: ${name}")
    endif()
    set(path "${TRIAEVUM_TRANSLATED_TITLE_DIR}/${name}")
    string(JSON expected GET "${metadata}" files "${name}")
    file(SHA256 "${path}" actual)
    if(NOT actual STREQUAL expected)
        message(FATAL_ERROR "Translated source hash mismatch: ${name}")
    endif()
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
    if(name MATCHES "\\.cpp$")
        list(APPEND sources "${path}")
    endif()
endforeach()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${manifest}")
find_package(nlohmann_json REQUIRED)
include("${root}/tools/triaevum_release/cmake/TitleSupport.cmake")
triaevum_add_title_support("${root}" "${root}/tools/oot3d/native_a32_runtime/upstream")
target_compile_options(triaevum_title_whole_aot_support PRIVATE -O2 -ffp-model=strict)
add_library(triaevum_macos_title SHARED
    ${sources}
    "${root}/tools/oot3d/native_game_runtime/triaevum_title_whole_aot_plugin.cpp")
target_include_directories(triaevum_macos_title PRIVATE "${TRIAEVUM_TRANSLATED_TITLE_DIR}")
target_compile_definitions(triaevum_macos_title PRIVATE OOT3D_NATIVE_GENERATED_WHOLE_AOT=1)
target_compile_options(triaevum_macos_title PRIVATE -O2 -ffp-model=strict)
target_link_libraries(triaevum_macos_title PRIVATE triaevum_title_whole_aot_support)
set_target_properties(triaevum_macos_title PROPERTIES PREFIX "" OUTPUT_NAME "triaevum_title_aot")
